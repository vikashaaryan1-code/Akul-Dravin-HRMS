  ): Promise<AttritionPrediction[]> {

      `SELECT id FROM employees WHERE tenant_id = $1 AND exit_date IS NULL AND status = 'active' LIMIT 500`,
      [tenantId],

      employees.map((e) => this.predictAttritionRisk(e.id, tenantId)),
